=== Test API ===
curl -i http://localhost:8080/health

=== Create Notification ===
curl -i -X POST http://localhost:8080/api/notifications \
  -H "Content-Type: application/json" \
  -d '{
    "recipient": "user@example.com",
    "subject": "Order #123",
    "message": "Your order has been shipped",
    "channel": "email",
    "is_urgent": true
  }'

=== List Notifications ===
curl -i http://localhost:8080/api/notifications

=== Get Notification ===
curl -i http://localhost:8080/api/notifications/1
